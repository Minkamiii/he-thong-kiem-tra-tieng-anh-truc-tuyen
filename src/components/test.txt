import React from "react";
import { useNavigate } from "react-router-dom";
import type { Test } from "../api/QuestionApi";

interface Props {
  test: Test;
}

const TopicCard: React.FC<Props> = ({ test }) => {
  const navigate = useNavigate();

  return (
    <div className="p-4 border rounded shadow bg-white">
      <h2 className="font-bold text-lg">{test.testName}</h2>
      <p className="text-gray-600">Type: {test.type}</p>
      <button
        className="mt-2 px-3 py-1 bg-blue-500 text-white rounded"
        onClick={() => navigate("/reading/detail", { state: { test } })}
      >
        Chi tiết
      </button>
    </div>
  );
};

export default TopicCard;
